_db_lock = threading.RLock()

def get_db():
    conn = sqlite3.connect(DB_FILE, check_same_thread=False, timeout=10)
    conn.row_factory = sqlite3.Row
    return conn

def init_db():
    with _db_lock:
        conn = get_db()
        conn.executescript('''
            CREATE TABLE IF NOT EXISTS trades (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                symbol TEXT, side TEXT, strategy TEXT,
                entry REAL, sl REAL, tp REAL, exit_price REAL,
                qty REAL, pnl REAL DEFAULT 0, pnl_pct REAL DEFAULT 0,
                score INTEGER, opened_at TEXT, closed_at TEXT,
                status TEXT DEFAULT 'open', leverage INTEGER DEFAULT 5,
                notes TEXT
            );
            CREATE TABLE IF NOT EXISTS logs (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                ts TEXT, level TEXT, message TEXT
            );
            CREATE TABLE IF NOT EXISTS predictions (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                ts TEXT, symbol TEXT, signal TEXT, confidence REAL, expected_return REAL
            );
        ''')
        conn.commit()
        conn.close()

def db_add_trade(symbol, side, strategy, entry, sl, tp, qty, score, leverage):
    with _db_lock:
        conn = get_db()
        cur = conn.execute('''INSERT INTO trades
            (symbol, side, strategy, entry, sl, tp, qty, score, opened_at, status, leverage)
            VALUES (?,?,?,?,?,?,?,?,?,?,?)''',
            (symbol, side, strategy, entry, sl, tp, qty, score,
             datetime.now().isoformat(), 'open', leverage))
        tid = cur.lastrowid
        conn.commit()
        conn.close()
        return tid

def db_close_trade(trade_id, exit_price, pnl, pnl_pct, notes=''):
    with _db_lock:
        conn = get_db()
        conn.execute('''UPDATE trades
            SET status='closed', exit_price=?, pnl=?, pnl_pct=?, closed_at=?, notes=?
            WHERE id=?''',
            (exit_price, pnl, pnl_pct, datetime.now().isoformat(), notes, trade_id))
        conn.commit()
        conn.close()

def db_get_trades(limit=50, status=None):
    with _db_lock:
        conn = get_db()
        if status:
            rows = conn.execute(
                'SELECT * FROM trades WHERE status=? ORDER BY id DESC LIMIT ?',
                (status, limit)).fetchall()
        else:
            rows = conn.execute(
                'SELECT * FROM trades ORDER BY id DESC LIMIT ?', (limit,)).fetchall()
        conn.close()
        return [dict(r) for r in rows]

def db_add_log(level, msg):
    with _db_lock:
        try:
            conn = get_db()
            conn.execute('INSERT INTO logs (ts, level, message) VALUES (?,?,?)',
                (datetime.now().isoformat(), level, msg))
            conn.commit()
            conn.close()
        except Exception:
            pass

def db_get_logs(limit=100):
    with _db_lock:
        conn = get_db()
        rows = conn.execute(
            'SELECT * FROM logs ORDER BY id DESC LIMIT ?', (limit,)).fetchall()
        conn.close()
        return [dict(r) for r in rows][::-1]

def db_add_prediction(symbol, signal, confidence, expected):
    with _db_lock:
        conn = get_db()
        conn.execute('''INSERT INTO predictions
            (ts, symbol, signal, confidence, expected_return)
            VALUES (?,?,?,?,?)''',
            (datetime.now().isoformat(), symbol, signal, confidence, expected))
        conn.commit()
        conn.close()

def db_get_stats():
    with _db_lock:
        conn = get_db()
        total = conn.execute('SELECT COUNT(*) FROM trades').fetchone()[0]
        wins = conn.execute("SELECT COUNT(*) FROM trades WHERE pnl>0").fetchone()[0]
        losses = conn.execute("SELECT COUNT(*) FROM trades WHERE pnl<0").fetchone()[0]
        total_pnl = conn.execute(
            "SELECT COALESCE(SUM(pnl),0) FROM trades WHERE status='closed'").fetchone()[0]
        conn.close()
        return {
            'total': total, 'wins': wins, 'losses': losses,
            'winrate': (wins/total*100) if total else 0,
            'total_pnl': total_pnl
        }

class FeatureEngineer:
    def build_features(self, df_15m, df_1h, df_4h):
        f = {}
        try:
            closes = df_15m['c'].values.astype(float)
            highs = df_15m['h'].values.astype(float)
            lows = df_15m['l'].values.astype(float)
            opens = df_15m['o'].values.astype(float)
            vols = df_15m['v'].values.astype(float)
        except Exception as e:
            log.error(f"Feature build: мэдээлэл буруу {e}")
            return {}

        for n in [1, 3, 5, 10, 20]:
            if len(closes) > n:
                f[f'ret_{n}'] = (closes[-1] - closes[-n-1]) / closes[-n-1]
            else:
                f[f'ret_{n}'] = 0

        f['volatility_20'] = pd.Series(closes[-20:]).pct_change().std() if len(closes) > 20 else 0
        f['hl_range'] = (highs[-1] - lows[-1]) / closes[-1] if closes[-1] else 0
        f['body_ratio'] = abs(closes[-1] - opens[-1]) / (highs[-1] - lows[-1] + 1e-9)
        f['upper_wick'] = (highs[-1] - max(closes[-1], opens[-1])) / closes[-1]
        f['lower_wick'] = (min(closes[-1], opens[-1]) - lows[-1]) / closes[-1]

        try:
            import pandas_ta as ta
            df = df_15m.copy()
            df['EMA9'] = ta.ema(df['c'], length=9)
            df['EMA21'] = ta.ema(df['c'], length=21)
            df['EMA50'] = ta.ema(df['c'], length=50)
            df['RSI'] = ta.rsi(df['c'], length=14)
            df['ATR'] = ta.atr(df['h'], df['l'], df['c'], length=14)

            macd = ta.macd(df['c'], fast=12, slow=26, signal=9)
            if macd is not None:
                df = pd.concat([df, macd], axis=1)

            bb = ta.bbands(df['c'], length=20, std=2)
            if bb is not None:
                df = pd.concat([df, bb], axis=1)

            adx = ta.adx(df['h'], df['l'], df['c'], length=14)
            if adx is not None:
                df['ADX_14'] = adx['ADX_14']

            stoch = ta.stoch(df['h'], df['l'], df['c'])
            if stoch is not None:
                df = pd.concat([df, stoch], axis=1)

            last = df.iloc[-1]
            f['rsi'] = last.get('RSI', 50) or 50
            f['atr_pct'] = (last.get('ATR', 0) or 0) / closes[-1]
            f['ema9_21_diff'] = ((last['EMA9'] or 0) - (last['EMA21'] or 0)) / closes[-1]
            f['price_ema21_diff'] = (closes[-1] - (last['EMA21'] or closes[-1])) / closes[-1]
            f['macd_hist'] = last.get('MACDh_12_26_9', 0) or 0
            f['adx'] = last.get('ADX_14', 0) or 0
            f['stoch_k'] = last.get('STOCHk_14_3_3', 50) or 50

            if 'BBU_20_2.0' in df.columns and 'BBL_20_2.0' in df.columns:
                bbu = last.get('BBU_20_2.0') or closes[-1]
                bbl = last.get('BBL_20_2.0') or closes[-1]
                f['bb_position'] = (closes[-1] - bbl) / (bbu - bbl + 1e-9)
            else:
                f['bb_position'] = 0.5
        except Exception as e:
            log.warning(f"pandas_ta алдаа: {e}")
            for k in ['rsi', 'atr_pct', 'ema9_21_diff', 'price_ema21_diff',
                     'macd_hist', 'adx', 'stoch_k', 'bb_position']:
                f[k] = 0

        if len(vols) > 20:
            f['vol_ratio'] = vols[-1] / (np.mean(vols[-20:]) + 1e-9)
        else:
            f['vol_ratio'] = 1

        if df_1h is not None and len(df_1h) > 50:
            try:
                import pandas_ta as ta
                df1 = df_1h.copy()
                df1['EMA50'] = ta.ema(df1['c'], length=50)
                df1['EMA200'] = ta.ema(df1['c'], length=200)
                f['1h_ema_diff'] = ((df1['EMA50'].iloc[-1] or 0) -
                                    (df1['EMA200'].iloc[-1] or 0)) / df1['c'].iloc[-1]
                f['1h_rsi'] = (ta.rsi(df1['c'], length=14).iloc[-1] or 50)
            except Exception:
                f['1h_ema_diff'] = 0
                f['1h_rsi'] = 50
        else:
            f['1h_ema_diff'] = 0
            f['1h_rsi'] = 50

        now = datetime.utcnow()
        f['hour_sin'] = np.sin(2 * np.pi * now.hour / 24)
        f['hour_cos'] = np.cos(2 * np.pi * now.hour / 24)

        return f


class AIStrategy:
    def __init__(self):
        self.model = None
        self.feature_names = None
        self.is_trained = False
        self.accuracy = 0.0
        self.total_predictions = 0
        self.correct_predictions = 0
        self.recent_accuracy = deque(maxlen=100)

    def train(self, symbol='BTC/USDT:USDT', days=30):
        log.info(f"📚 AI сургалт эхлэв: {symbol} ({days} хоног)")
        try:
            from sklearn.ensemble import RandomForestClassifier
            import pandas_ta as ta
            import joblib

            ex = ccxt.binance({'enableRateLimit': True})
            since = int((datetime.now() - timedelta(days=days)).timestamp() * 1000)
            bars = []
            log.info(f"📥 {symbol} мэдээлэл татаж байна...")

            while True:
                b = ex.fetch_ohlcv(symbol, '15m', since=since, limit=1000)
                if not b:
                    break
                bars.extend(b)
                since = b[-1][0] + 1
                if len(b) < 1000:
                    break
                if len(bars) > 5000:
                    break

            if len(bars) < 500:
                log.error(f"Мэдээлэл хангалтгүй: {len(bars)} баар")
                return False

            log.info(f"✅ {len(bars)} баар татлаа")

            df = pd.DataFrame(bars, columns=['t', 'o', 'h', 'l', 'c', 'v'])
            df['t'] = pd.to_datetime(df['t'], unit='ms')

            # Scalp шошго (2 баар, 0.1%)
            df['future'] = df['c'].shift(-2)
            df['ret'] = (df['future'] - df['c']) / df['c']
            df['label'] = 0
            df.loc[df['ret'] > 0.001, 'label'] = 1
            df.loc[df['ret'] < -0.001, 'label'] = 2

            try:
                bars_1h = ex.fetch_ohlcv(symbol, '1h', limit=500)
                df_1h = pd.DataFrame(bars_1h, columns=['t', 'o', 'h', 'l', 'c', 'v'])
            except Exception:
                df_1h = None

            fe = FeatureEngineer()
            X_list, y_list = [], []
            log.info("🔧 Шинж чанар тооцоолж байна...")

            for i in range(60, len(df) - 4):
                if df.iloc[i]['label'] == 0 and (i % 3 != 0):
                    continue
                try:
                    s15 = df.iloc[max(0, i-100):i+1].reset_index(drop=True)
                    if df_1h is not None:
                        s1h = df_1h[df_1h['t'] <= df.iloc[i]['t']].tail(100).reset_index(drop=True)
                    else:
                        s1h = None
                    feats = fe.build_features(s15, s1h, None)
                    if feats:
                        X_list.append(feats)
                        y_list.append(int(df.iloc[i]['label']))
                except Exception:
                    continue

            if len(X_list) < 100:
                log.error(f"Шинж чанар хангалтгүй: {len(X_list)}")
                return False

            X = pd.DataFrame(X_list).fillna(0)
            y = np.array(y_list)
            self.feature_names = list(X.columns)

            log.info(f"✅ {len(X)} сургалтын дээж, {len(self.feature_names)} шинж чанар")

            split = int(len(X) * 0.8)
            X_train, X_test = X.iloc[:split], X.iloc[split:]
            y_train, y_test = y[:split], y[split:]

            self.model = RandomForestClassifier(
                n_estimators=200, max_depth=10,
                min_samples_leaf=20, class_weight='balanced',
                random_state=42, n_jobs=-1
            )
            self.model.fit(X_train, y_train)
            acc = (self.model.predict(X_test) == y_test).mean()
            self.accuracy = acc
            self.is_trained = True

            joblib.dump({'model': self.model, 'features': self.feature_names}, 'ai_model.pkl')
            log.info(f"✅ AI сурлаа: accuracy={acc*100:.1f}%, {len(X)} дээж")
            return True
        except Exception as e:
            log.error(f"Train алдаа: {e}")
            import traceback
            traceback.print_exc()
            return False

    def predict(self, df_15m, df_1h, df_4h):
        if not self.is_trained or self.model is None:
            return None
        try:
            fe = FeatureEngineer()
            f = fe.build_features(df_15m, df_1h, df_4h)
            if not f:
                return None

            for c in self.feature_names:
                if c not in f:
                    f[c] = 0

            X = pd.DataFrame([f])[self.feature_names].fillna(0)
            proba = self.model.predict_proba(X)[0]
            pred = int(np.argmax(proba))
            conf = float(proba[pred])

            classes = self.model.classes_
            proba_dict = {int(c): float(p) for c, p in zip(classes, proba)}

            return {
                'signal': ['skip', 'long', 'short'][pred] if pred < 3 else 'skip',
                'confidence': conf,
                'proba': {
                    'skip': proba_dict.get(0, 0),
                    'long': proba_dict.get(1, 0),
                    'short': proba_dict.get(2, 0)
                }
            }
        except Exception as e:
            log.error(f"Predict алдаа: {e}")
            return None

    def load(self):
        try:
            import joblib
            if Path('ai_model.pkl').exists():
                data = joblib.load('ai_model.pkl')
                self.model = data['model']
                self.feature_names = data['features']
                self.is_trained = True
                log.info("✅ AI model ачаалагдлаа")
                return True
        except Exception as e:
            log.error(f"Model ачаалахад алдаа: {e}")
        return False

    def record_result(self, predicted_class, actual_class):
        self.total_predictions += 1
        if predicted_class == actual_class:
            self.correct_predictions += 1
            self.recent_accuracy.append(1)
        else:
            self.recent_accuracy.append(0)

    def get_recent_accuracy(self):
        return sum(self.recent_accuracy) / len(self.recent_accuracy) if self.recent_accuracy else 0

    def __init__(self):

    def emit(self, et, data):

def get_ai_signal(symbol):
    if not AI.is_trained:
        return None
    df_15m = fetch_ohlcv(symbol, '15m', 100)
    df_1h = fetch_ohlcv(symbol, '1h', 100)
    df_4h = fetch_ohlcv(symbol, '4h', 50)
    if df_15m is None or len(df_15m) < 60:
        return None

    pred = AI.predict(df_15m, df_1h, df_4h)
    if pred is None or pred['signal'] == 'skip':
        return None

    if pred['confidence'] < STATE.config.get('AI_CONFIDENCE_THRESHOLD', 0.62):
        return None

    try:
        import pandas_ta as ta
        atr = ta.atr(df_15m['h'], df_15m['l'], df_15m['c'], length=14).iloc[-1]
        if pd.isna(atr) or atr <= 0:
            return None
    except Exception:
        return None

    entry = df_15m['c'].iloc[-1]
    side = 'buy' if pred['signal'] == 'long' else 'sell'

    if STATE.config.get('SCALP_MODE'):
        sl_mult = 0.8
        tp_mult = 1.2
    else:
        sl_mult = 1.2
        tp_mult = 2.4

    if side == 'buy':
        sl = entry - atr * sl_mult
        tp = entry + atr * tp_mult
    else:
        sl = entry + atr * sl_mult
        tp = entry - atr * tp_mult

    return {
        'symbol': symbol, 'side': side,
        'strategy': f"ai_{pred['signal']}",
        'score': int(pred['confidence'] * 100),
        'entry': entry, 'sl': sl, 'tp': tp,
        'confidence': pred['confidence'],
        'proba': pred['proba']
    }

def calc_qty(symbol, entry, sl, balance, leverage):
    try:
        max_use = STATE.config.get('MAX_BALANCE_USE', 50)
        usable_balance = min(balance, max_use)

        market = STATE.exchange.market(symbol)
        step = market['precision']['amount'] or 0.001
        min_qty = market['limits']['amount']['min'] or step
        rpc = abs(entry - sl)
        if rpc <= 0:
            return None
        risk_amount = usable_balance * STATE.config.get('MAX_RISK_PER_TRADE', 0.015)
        qty = risk_amount / rpc

        max_notional = usable_balance * leverage
        if qty * entry > max_notional:
            qty = max_notional / entry

        qty = max(min_qty, (qty // step) * step)
        return round(qty, 8) if qty >= min_qty else None
    except Exception as e:
        log.error(f"calc_qty: {e}")
        return None

def close_position(symbol, reason='manual'):
    if symbol not in STATE.positions:
        return 0.0
    pos = STATE.positions[symbol]
    try:
        positions = STATE.exchange.fetch_positions([symbol])
        amt = 0.0
        for p in positions:
            amt = float(p['info'].get('positionAmt', 0))
            if amt != 0:
                break
        if amt != 0:
            side = 'sell' if amt > 0 else 'buy'
            STATE.exchange.create_market_order(
                symbol, side, abs(amt),
                params={'reduceOnly': True}
            )
        for _ in range(10):
            time.sleep(1)
            positions = STATE.exchange.fetch_positions([symbol])
            amt = sum(abs(float(p['info'].get('positionAmt', 0))) for p in positions)
            if amt == 0:
                break
    except Exception as e:
        log.error(f"close: {e}")
        return 0.0

    new_bal = get_balance() or 0
    pnl = new_bal - pos['balance_before']
    pnl_pct = (pnl / pos['balance_before'] * 100) if pos['balance_before'] else 0

    db_close_trade(pos['trade_id'], pos['entry'], pnl, pnl_pct, reason)

    if pnl > 0:
        STATE.wins += 1
        STATE.consecutive_wins += 1
        STATE.consecutive_losses = 0
    elif pnl < 0:
        STATE.losses += 1
        STATE.consecutive_losses += 1
        STATE.consecutive_wins = 0

    actual = 1 if pnl > 0 else 2 if pnl < 0 else 0
    if pos.get('predicted_class') is not None:
        AI.record_result(pos['predicted_class'], actual)

    STATE.daily_pnl += pnl
    STATE.last_trade_close_time = time.time()
    STATE.positions.pop(symbol, None)
    STATE.emit('position_closed', {
        'symbol': symbol, 'pnl': pnl, 'reason': reason
    })

    emoji = '✅' if pnl > 0 else '❌'
    STATE.add_log('INFO', f"{emoji} {symbol} {reason}: ${pnl:+.3f}")
    send_telegram(f"{emoji} {symbol.split(':')[0]} PnL: ${pnl:+.3f}\n📝 {reason}")
    return pnl

def execute_trade(sig):
    cfg = STATE.config
    symbol = sig['symbol']
    bal = get_balance()
    if not bal or bal < cfg.get('MIN_BALANCE', 15):
        STATE.add_log('WARNING', f"Баланс {cfg.get('MIN_BALANCE',15)}$-аас бага: ${bal}")
        return False

    if cfg.get('USE_CLAUDE'):
        summary = build_features_summary(symbol)
        if not claude_confirm_signal(symbol, sig, summary):
            STATE.add_log('INFO', f"⏭️ {symbol}: Claude татгалзсан тул алгассан")
            return False

    lev = cfg.get('LEVERAGE', 5)
    lev += min(2, STATE.consecutive_wins // 3)
    lev -= min(2, STATE.consecutive_losses // 2)
    lev = max(3, min(7, lev))

    try:
        STATE.exchange.set_leverage(lev, symbol)
    except Exception:
        pass

    qty = calc_qty(symbol, sig['entry'], sig['sl'], bal, lev)
    if not qty:
        STATE.add_log('WARNING', f"qty тооцож чадсангүй: {symbol}")
        return False

    try:
        STATE.exchange.create_market_order(symbol, sig['side'], qty)
    except Exception as e:
        STATE.add_log('ERROR', f"Order алдаа: {e}")
        return False

    close_side = 'sell' if sig['side'] == 'buy' else 'buy'
    try:
        STATE.exchange.create_order(
            symbol, 'STOP_MARKET', close_side, qty,
            params={'stopPrice': sig['sl'], 'reduceOnly': True,
                    'workingType': 'MARK_PRICE'}
        )
        STATE.exchange.create_order(
            symbol, 'TAKE_PROFIT_MARKET', close_side, qty,
            params={'stopPrice': sig['tp'], 'reduceOnly': True,
                    'workingType': 'MARK_PRICE'}
        )
    except Exception as e:
        STATE.add_log('ERROR', f"SL/TP алдаа: {e}")

    pred_class = 1 if sig['side'] == 'buy' else 2
    trade_id = db_add_trade(
        symbol, sig['side'], sig['strategy'], sig['entry'],
        sig['sl'], sig['tp'], qty, sig['score'], lev
    )
    db_add_prediction(symbol, sig['side'], sig['confidence'], sig.get('expected_return', 0))

    STATE.positions[symbol] = {
        'side': sig['side'], 'qty': qty, 'entry': sig['entry'],
        'sl': sig['sl'], 'tp': sig['tp'], 'trade_id': trade_id,
        'strategy': sig['strategy'], 'score': sig['score'],
        'confidence': sig['confidence'], 'predicted_class': pred_class,
        'balance_before': bal, 'leverage': lev,
        'opened_at': datetime.now().isoformat()
    }
    STATE.trades_today += 1
    STATE.total_trades += 1
    STATE.emit('position_opened', STATE.positions[symbol])

    side_lbl = '🟢 LONG' if sig['side'] == 'buy' else '🔴 SHORT'
    msg = (f"{side_lbl} <b>{symbol.split(':')[0]}</b>\n"
           f"🧠 AI: {sig['confidence']*100:.0f}%\n⚡ Lev: {lev}x\n"
           f"💰 {qty}\n📍 {sig['entry']:.4f}\n"
           f"🛑 {sig['sl']:.4f}\n🎯 {sig['tp']:.4f}")
    send_telegram(msg)
    STATE.add_log('INFO', f"{side_lbl} {symbol} @ {sig['entry']:.4f}")
    return True

# 9. TRAILING STOP

def run_trailing(symbol):
    if not STATE.config.get('USE_TRAILING'):
        return
    pos = STATE.positions.get(symbol)
    if not pos:
        return
    cfg = STATE.config
    triggered = False
    stop = pos['entry'] * (0.992 if pos['side'] == 'buy' else 1.008)

    while symbol in STATE.positions and STATE.running:
        try:
            cur = STATE.exchange.fetch_ticker(symbol)['last']
            if pos['side'] == 'buy':
                if not triggered and cur >= pos['entry'] * (1 + cfg['TRAILING_TRIGGER']):
                    triggered = True
                    stop = cur * (1 - cfg['TRAILING_DISTANCE'])
                if triggered:
                    stop = max(stop, cur * (1 - cfg['TRAILING_DISTANCE']))
                    if cur <= stop:
                        close_position(symbol, 'trailing')
                        break
            else:
                if not triggered and cur <= pos['entry'] * (1 - cfg['TRAILING_TRIGGER']):
                    triggered = True
                    stop = cur * (1 + cfg['TRAILING_DISTANCE'])
                if triggered:
                    stop = min(stop, cur * (1 + cfg['TRAILING_DISTANCE']))
                    if cur >= stop:
                        close_position(symbol, 'trailing')
                        break
        except Exception:
            pass
        time.sleep(2)

def trainer_loop():
    interval = STATE.config.get('AI_RETRAIN_INTERVAL', 7200)
    symbol = STATE.config['SYMBOLS'][0] if STATE.config.get('SYMBOLS') else 'BTC/USDT:USDT'
    while STATE.running:
        if STATE.last_train is None or \
           (datetime.now() - STATE.last_train).total_seconds() >= interval:
            try:
                STATE.add_log('INFO', f"🔄 Auto-retrain эхлэв ({symbol})")
                if AI.train(symbol, days=14):
                    STATE.last_train = datetime.now()
                    STATE.emit('ai_trained', {'accuracy': AI.accuracy})
            except Exception as e:
                STATE.add_log('ERROR', f"Auto-retrain: {e}")
        time.sleep(60)

def bot_loop():
    cfg = STATE.config
    STATE.start_time = datetime.now()

    if not AI.is_trained:
        if not AI.load():
            STATE.add_log('INFO', '🧠 AI model байхгүй. Сургаж байна... (5-10 мин)')
            sym = cfg['SYMBOLS'][0] if cfg.get('SYMBOLS') else 'BTC/USDT:USDT'
            if not AI.train(sym, days=30):
                STATE.add_log('ERROR', '❌ AI сургалт амжилтгүй.')
                STATE.running = False
                return
        STATE.last_train = datetime.now()

    if STATE.trainer_thread is None or not STATE.trainer_thread.is_alive():
        STATE.trainer_thread = threading.Thread(target=trainer_loop, daemon=True)
        STATE.trainer_thread.start()

    try:
        init_exchange()
    except Exception as e:
        STATE.add_log('ERROR', f"Exchange холболт: {e}")
        STATE.running = False
        return

    bal = get_balance()
    if not bal:
        STATE.add_log('ERROR', "Баланс авах боломжгүй — API key шалгана уу")
        STATE.running = False
        return

    STATE.initial_balance = bal
    STATE.peak_balance = bal
    STATE.last_balance = bal
    claude_status = "🤖 Claude: ON" if cfg.get('USE_CLAUDE') else "🤖 Claude: OFF"
    send_telegram(
        f"🧠💰 AI Bot эхэллээ\n"
        f"💵 ${bal:.2f}\n"
        f"📊 Accuracy: {AI.get_recent_accuracy()*100:.1f}%\n"
        f"{claude_status}"
    )
    STATE.add_log('INFO', f"🧠 Bot аслаа. ${bal:.2f} | {claude_status}")

    last_emit = 0
    while STATE.running:
        try:
            if date.today() != STATE.last_day_reset:
                STATE.trades_today = 0
                STATE.last_day_reset = date.today()
                STATE.daily_pnl = 0

            cur_bal = get_balance() or 0
            if cur_bal > 0:
                if STATE.daily_pnl >= cur_bal * cfg['DAILY_PROFIT_TARGET']:
                    STATE.add_log('INFO', f"🎯 Daily profit target хүрлээ")
                    send_telegram("🎯 Өдрийн profit target! Зогсож байна")
                    STATE.running = False
                    break
                if STATE.daily_pnl <= -cur_bal * cfg['DAILY_LOSS_LIMIT']:
                    STATE.add_log('WARNING', f"🛑 Daily loss limit хүрлээ")
                    send_telegram("🛑 Өдрийн loss limit! Зогсож байна")
                    STATE.running = False
                    break

            for sym in list(STATE.positions.keys()):
                t = STATE.trailing_threads.get(sym)
                if not t or not t.is_alive():
                    STATE.trailing_threads[sym] = threading.Thread(
                        target=run_trailing, args=(sym,), daemon=True
                    )
                    STATE.trailing_threads[sym].start()

            for sym in list(STATE.positions.keys()):
                try:
                    positions = STATE.exchange.fetch_positions([sym])
                    amt = sum(abs(float(p['info'].get('positionAmt', 0))) for p in positions)
                    if amt == 0:
                        close_position(sym, 'sl_or_tp')
                except Exception:
                    pass

            if (len(STATE.positions) < cfg['MAX_POSITIONS'] and
                STATE.trades_today < cfg['MAX_TRADES_PER_DAY'] and
                time.time() - STATE.last_trade_close_time >= cfg['GLOBAL_COOLDOWN']):

                best_sig = None
                for sym in cfg.get('SYMBOLS', []):
                    if sym in STATE.positions:
                        continue
                    sig = get_ai_signal(sym)
                    if sig and (not best_sig or sig['confidence'] > best_sig['confidence']):
                        best_sig = sig

                if best_sig:
                    execute_trade(best_sig)

            now = time.time()
            if now - last_emit >= 2:
                STATE.emit('tick', get_snapshot())
                last_emit = now

            time.sleep(2)
        except Exception as e:
            STATE.add_log('ERROR', f"Loop алдаа: {e}")
            time.sleep(5)

    STATE.add_log('WARNING', '⏹️ Bot зогслоо')
    send_telegram("⏹️ Bot зогслоо")

def start_bot():
    if STATE.running:
        return False, "Аль хэдийн ажиллаж байна"
    if not STATE.config.get('API_KEY') and not STATE.config.get('TESTNET'):
        return False, "API түлхүүр оруулаагүй байна"
    STATE.running = True
    STATE.thread = threading.Thread(target=bot_loop, daemon=True)
    STATE.thread.start()
    return True, "🧠 Bot эхэллээ"

def stop_bot():
    STATE.running = False
    return True, "Зогсох хүсэлт илгээгдлээ"

def require_login(f):
    from functools import wraps
    @wraps(f)
    def wrapped(*args, **kwargs):
        if not session.get('logged_in'):
            return jsonify({'error': 'login required'}), 401
        return f(*args, **kwargs)
    return wrapped

@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        if check_login(request.form.get('username', ''), request.form.get('password', '')):
            session['logged_in'] = True
            session['username'] = request.form.get('username')
            return redirect(url_for('dashboard'))
        return render_template_string(HTML_TEMPLATE, logged_in=False, error='Буруу хэрэглэгч эсвэл нууц үг')
    return render_template_string(HTML_TEMPLATE, logged_in=False)

@app.route('/logout')
def logout():
    session.clear()
    return redirect(url_for('login'))

@app.route('/')
def dashboard():
    if not session.get('logged_in'):
        return redirect(url_for('login'))
    return render_template_string(HTML_TEMPLATE, logged_in=True)

@require_login
def api_config():
    if request.method == 'GET':
        cfg = load_config()
        cfg['API_KEY_SET'] = bool(cfg.get('API_KEY'))
        cfg.pop('API_KEY', None)
        cfg.pop('API_SECRET', None)
        cfg.pop('CLAUDE_API_KEY', None)
        return jsonify(cfg)
    try:
        data = request.json
        if not data:
            return jsonify({'ok': False, 'msg': 'Өгөгдөл байхгүй'}), 400
        cfg = load_config()
        for k, v in data.items():
            if k in ('API_KEY', 'API_SECRET', 'CLAUDE_API_KEY') and not v:
                continue
            if k in DEFAULT_CONFIG or k in ('TELEGRAM_TOKEN', 'CHAT_ID'):
                cfg[k] = v
        save_config(cfg)
        STATE.config = cfg
        return jsonify({'ok': True, 'msg': 'Хадгалагдлаа'})
    except Exception as e:
        return jsonify({'ok': False, 'msg': str(e)}), 500

@app.route('/api/ai/train', methods=['POST'])
@require_login
def api_ai_train():
    def task():
        STATE.add_log('INFO', '📚 AI сургалт эхлэв...')
        sym = STATE.config['SYMBOLS'][0] if STATE.config.get('SYMBOLS') else 'BTC/USDT:USDT'
        ok = AI.train(sym, days=30)
        if ok:
            STATE.last_train = datetime.now()
            STATE.emit('ai_trained', {'accuracy': AI.accuracy})
            STATE.add_log('INFO', f'✅ AI сурлаа: {AI.accuracy*100:.1f}%')
        else:
            STATE.add_log('ERROR', '❌ AI сургалт амжилтгүй')
    threading.Thread(target=task, daemon=True).start()
    return jsonify({'ok': True})
